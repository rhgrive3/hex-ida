function stackReturnFixture({ includeLoad = true, storeSize = 4, barrier = false } = {}) {
  const key = 'stack:sp:e0:-16:s4';
  const value = { id: 100 };
  const store = {
    id: 10, op: 'store', block: 0, row: 0, address: 0x1000n,
    loc: { kind: 'stack', key, size: storeSize }, args: [{ value }],
  };
  const load = {
    id: 11, op: 'load', block: 0, row: 2, address: 0x1008n,
    loc: { kind: 'stack', key, size: 4 }, args: [],
  };
  const unknown = { id: 12, op: 'unknown', block: 0, row: 1, address: 0x1004n, args: [] };
  const ret = { id: 13, op: 'ret', block: 0, row: 3, address: 0x100cn, args: [] };
  const instructions = [store, ...(barrier ? [unknown] : []), ...(includeLoad ? [load] : []), ret];
  const expression = expr.load({ kind: 'stack', key }, 32, { row: load.row, address: load.address, ir: load.id });
  return {
    semantic: true,
    ir: { instructions, blocks: [{ index: 0, startRow: 0, endRow: 3, pred: [], succ: [], insts: instructions }], idom: [-1] },
    semanticAst: {
      values: [{ valueId: value.id, expression: expr.constant(11n, 32, true) }],
      conditions: [],
      outputs: [{ name: 'return', expression }],
    },
    cAst: { body: [{
      kind: 'stmt', indent: 0, text: 'return local_0;',
      semantic: { op: 'return', expression },
      source: { rows: [ret.row], addresses: [ret.address], ir: [ret.id] },
    }] },
    rewriteProof: [],
    metrics: { rewrittenExpressions: 0 },
  };
}

function phase8IdentityFixture(name) {
  const f = fixture(name);
  f.block(0);
  const input = f.opaque(8);
  f.binary('add', input, f.constant(1, 8), 8);
  f.ret();
  return f.build();
}

function phase8GvnFixture(name) {
  const f = fixture(name);
  f.block(0);
  const left = f.opaque(32);
  const right = f.opaque(32);
  const first = f.binary('add', left, right, 32);
  const second = f.binary('add', left, right, 32);
  f.ret();
  return { ir: f.build(), first, second };
}

function solverQuery(assertion, constraints = []) {
  return createVerificationQuery({
    kind: VERIFICATION_QUERY_KIND.CONDITIONAL_EDGE_FEASIBILITY,
    claimKind: CLAIM_KIND.EDGE_FEASIBLE,
    targetEntity: 't019-recovery-probe',
    constraints,
    assertion,
  });
}

function discover({ binaryId = 't019-binary', sourceHash = 'source-v1', sizeBytes } = {}) {
  const functionRecord = { address: 0x1000, source: 'function_starts' };
  if (sizeBytes !== undefined) functionRecord.sizeBytes = sizeBytes;
  return functionCandidates({
    input: { image: { functions: [functionRecord] } },
    binaryId,
    sourceHash,
    snapshotId: 'snapshot-v1',
    architectureId: 'x86_64',
  });
}

function dyldHeader({ mappingCount = 0, mappingOffset = 104 } = {}) {
  const bytes = new Uint8Array(104);
  const magic = new TextEncoder().encode('dyld_v1  x86_64');
  bytes.set(magic, 0);
  const view = new DataView(bytes.buffer);
  view.setUint32(16, mappingOffset, true);
  view.setUint32(20, mappingCount, true);
  return bytes;
}

function minimalMachO() {
  const bytes = new Uint8Array(32);
  const view = new DataView(bytes.buffer);
  view.setUint32(0, 0xfeedfacf, true);
  view.setInt32(4, 0x01000007, true);
  view.setInt32(8, 3, true);
  view.setUint32(12, 2, true);
  view.setUint32(16, 0, true);
  view.setUint32(20, 0, true);
  view.setUint32(24, 0, true);
  view.setUint32(28, 0, true);
  return bytes;
}

function valueOperation(undefinedResult) {
  return createMachineOperation({
    kind: 'value', opcode: 't019-boundary',
    inputs: [createBitVectorValue(32, 1n)],
    outputs: [createTemporaryValue('t019-output', createBitVectorValue(32))],
    undefinedResult,
  });
}

for (const { lane, cases } of lanes) {
  let stopped = false;
  for (const [id, run] of cases) {
    if (stopped) {
      console.log(JSON.stringify({ lane, id, status: 'not-run-after-first-failure' }));
      continue;
    }
    const started = Date.now();
    try {
      await run();
      console.log(JSON.stringify({ lane, id, status: 'passed', elapsedMs: Date.now() - started }));
    } catch (error) {
      stopped = true;
      console.log(JSON.stringify({
        lane, id, status: 'failed', elapsedMs: Date.now() - started,
        error: { name: error?.name, message: String(error?.message || error), stack: error?.stack },
      }));
    }
  }
}
